 262      private void handleNavigation(Intent intent) {
 263          if (intent == null || navController == null) {
 264              return;
 265          }
 266          
 267          String navigateTo = intent.getStringExtra("navigate_to");
 268          int destinationId;
 269          
 270          if ("documents".equals(navigateTo)) {
 271              // 从外部打开PDF后返回，导航到文档列表
 272              destinationId = R.id.nav_documents;
 273          } else if ("home".equals(navigateTo)) {
 274              // 从外部打开WebView后返回，导航到首页
 275              destinationId = R.id.nav_home;
 276          } else {
 277              // 读取默认Tab设置并导航
 278              SharedPreferences prefs = getPreferences(Context.MODE_PRIVATE);
 279              int defaultTab = prefs.getInt("default_tab", 0); // 默认为0，即首页
 280              
 281              // 根据设置选择目标页面ID
 282              switch (defaultTab) {
 283                  case 0: // 首页
 284                      destinationId = R.id.nav_home;
 285                      break;
 286                  case 2: // 主题
 287                      destinationId = R.id.nav_subject;
 288                      break;
 289                  case 3: // RSS
 290                      destinationId = R.id.nav_rss;
 291                      break;
 292                  case 4: { // 随机：首页/主题/RSS
 293                      int r = (int) (Math.random() * 3);
 294                      destinationId = (r == 0)
 295                          ? R.id.nav_home
 296                          : (r == 1 ? R.id.nav_subject : R.id.nav_rss);
 297                      break;
 298                  }
 299                  default: // 默认首页（包括已废弃的case 1标签页）
 300                      destinationId = R.id.nav_home;
 301                      break;
 302              }
 303          }
 304          
 305          // 导航到目标页面
 306          try {
 307              navController.navigate(destinationId);
 308          } catch (Exception e) {
 309              Log.e("MainActivity", "导航失败", e);
 310          }
 311      }

 414                          return;

 419                      }

 421                      String savedText = prefs.getString("last_clipboard_text", "" // <wrapped>
 422                      if (!clipTextSequence.toString().equals(savedText)) {

 424                          handleClipboardText(lastClipboardText); //@Def Line376

 455      private void handleClipboardText(String text) {
 456          if (TextUtils.isEmpty(text)) {
 457              return;
 458          }
 459  
 460          Log.d("LinkParser", "Processing text: " + text);
 461          
 462          // 查找第一个 http(s) URL
 463          Pattern urlPattern = Pattern.compile("https?://[^\\s,，]+");
 464          Matcher urlMatcher = urlPattern.matcher(text);
 465          
 466          if (urlMatcher.find()) {
 467              // 获取URL
 468              String url = urlMatcher.group(0);
 469              // 移除URL中可能的查询参数
 470              String cleanUrl = url.split("\\?")[0];
 471              // 检查是否以.xml结尾（忽略大小写）
 472              if (cleanUrl.toLowerCase().endsWith(".xml")) {
 473                  return;
 474              }
 475              
 476              // 获取URL之前的内容作为标题
 477              String title = text.substring(0, urlMatcher.start()).trim();
 478              
 479              // 如果标题为空或者太长，进行处理
 480              if (TextUtils.isEmpty(title)) {
 481                  title = "新链接";
 482              } else if (title.length() > 20) {
 483                  title = title.substring(0, 20) + "...";
 484              }
 485              
 486              Log.d("LinkParser", "Extracted title: " + title);
 487              Log.d("LinkParser", "Extracted URL: " + url);
 488  
 489              // 显示保存对话框
 490              final String finalTitle = title;
 491              final String finalUrl = url;
 492              new Handler(Looper.getMainLooper()).post(() -> {
 493                  showSaveLinkDialog(finalUrl, finalTitle);
 494              });
 495          }
 496      }
