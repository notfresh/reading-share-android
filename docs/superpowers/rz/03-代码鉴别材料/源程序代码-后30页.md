 269      private static String escapeCSV(String value) {
 270          if (value == null) return "";
 271          value = value.replace("\"", "\"\""); // 转义双引号
 272          if (value.contains(",") || value.contains("\"") || value.contains("\n")) // <wrapped>
 273              value = "\"" + value + "\""; // 如果包含逗号或换行，用双引号包围
 274          }
 275          return value;
