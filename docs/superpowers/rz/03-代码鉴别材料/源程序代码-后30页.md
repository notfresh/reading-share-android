 137          float scale = (float) size / Math.min(width, height);

 139          int scaledHeight = (int) (height * scale);

 147              int y = (scaledHeight - size) / 2;

 152

 158              }

 161              return squareBitmap;

 330      private void handleExternalPdf(Uri uri) {
 331          Log.d(TAG, "处理外部PDF: " + uri.toString());
 332  
 333          try {
 334              // 获取文件名
 335              String fileName = getFileName(uri);
 336              if (fileName == null || fileName.isEmpty()) {
 337                  fileName = "external_" + System.currentTimeMillis() + ".pdf";
 338              }
 339  
 340              // 复制文件到应用私有目录
 341              String destPath = copyFileToPrivateDir(uri, fileName);
 342              if (destPath == null) {
 343                  // 如果复制失败，尝试直接打开外部文件
 344                  pdfFile = getFileFromUri(uri);
 345                  if (pdfFile == null || !pdfFile.exists()) {
 346                      Toast.makeText(this, "无法打开PDF文件", Toast.LENGTH_SHORT).show() // <wrapped>
 347                      finish();

 398      private String getFileName(Uri uri) {
 399          String fileName = null;
 400          String scheme = uri.getScheme();
 401

 403              android.database.Cursor cursor = getContentResolver().query(
 404                      uri, null, null, null, null);
 405              if (cursor != null && cursor.moveToFirst()) {
 406                  int nameIndex = cursor.getColumnIndex(android.provider.OpenableC // <wrapped>

 408                      fileName = cursor.getString(nameIndex);
 409                  }
 410                  cursor.close();
 411              }
 412          } else if ("file".equals(scheme)) {
 413              String path = uri.getPath();
 414              if (path != null) {
 415                  int cut = path.lastIndexOf('/');
 416                  if (cut != -1) {
 417                      fileName = path.substring(cut + 1);
 418                  } else {
 419                      fileName = path;
 420                  }
 421              }
 422          }
 423          
 424          return fileName;
 425      }

 208                  getInstance().removeWebView(currentUrl);

 231      protected void onDestroy() {

 233              // 清理 WebView
 234              if (webView != null) {
 235                  webView.stopLoading();

 237                  webView.clearCache(true);
 238                  webView.destroy();
 239              }
 240          }
 241          if(preserveCache){  // 只生效一次,反转回来
 242             preserveCache = false;
 243          }
 244  
 245          // 无论是否保留缓存，都在onDestroy中释放这些资源
 246          stopService(new Intent(this, WebViewBackgroundService.class));
 247          WebViewBackgroundService.setMediaCallback(null);
 248          if (audioManager != null && audioFocusRequest != null) {
 249              audioManager.abandonAudioFocusRequest(audioFocusRequest);
 250          }
 251          if (mediaSession != null) {
 252              mediaSession.setActive(false);
 253              mediaSession.release();
 254              mediaSession = null;
 255          }
 256          if (wakeLock != null && wakeLock.isHeld()) {
 257              wakeLock.release();
 258              wakeLock = null;
 259          }
 260          super.onDestroy();
 261      }

  45          PowerManager pm = (PowerManager) getSystemService(Context.POWER_SERVICE) // <wrapped>

  59              if (ACTION_PLAY_PAUSE.equals(action)) {

  61                  Log.d(TAG, "Play/Pause 按钮点击, isPaused=" + isPaused + ", callback // <wrapped>
  62                  if (sMediaCallback != null) {

  65                      } else {

  67                      }
  68                  }
  69                  // 刷新通知
  70                  startForeground(NOTIFICATION_ID, buildNotification());
  71                  return START_STICKY;
  72              } else if (ACTION_STOP.equals(action)) {
  73                  Log.d(TAG, "Stop 按钮点击, callback=" + (sMediaCallback != null));
  74                  if (sMediaCallback != null) {
  75                      sMediaCallback.onStopRequested();

  85              String url = intent.getStringExtra("current_url");
  86              if (url != null) currentUrl = url;
